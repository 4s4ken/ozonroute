func (s *Service) Accept(ctx context.Context, order pvz.OrderCreate) error {
        now := s.nowFn().UTC()

        if order.ExpiresAt.Before(now) {
                return errors.EntityValidationError("Order", strconv.FormatUint(uint64(order.OrderID), 10))
        }

        existing, err := s.repo.GetOrder(ctx, order.OrderID)
        if err != nil {
                return err
        }
        if existing != nil {
                return errors.EntityAlreadyExistsError("Order", strconv.FormatUint(uint64(order.OrderID), 10))
        }

        finalPrice := order.Price

        if order.Package != nil {
                switch *order.Package {
                case pvz.PackageTypeBag:
                        if order.Weight > 10 {
                                return errors.EntityWeightError("Bag", strconv.FormatUint(uint64(order.OrderID), 10))
                        }
                        finalPrice = finalPrice.Add(decimal.NewFromInt(5))
                case pvz.PackageTypeBox:
                        if order.Weight > 30 {
                                return errors.EntityWeightError("Box", strconv.FormatUint(uint64(order.OrderID), 10))
                        }
                        finalPrice = finalPrice.Add(decimal.NewFromInt(20))
                case pvz.PackageTypeTape:
                        finalPrice = finalPrice.Add(decimal.NewFromInt(1))
                case pvz.PackageTypeBagTape:
                        if order.Weight > 10 {
                                return errors.EntityWeightError("Bag+Tape", strconv.FormatUint(uint64(order.OrderID), 10))
                        }
                        finalPrice = finalPrice.Add(decimal.NewFromInt(6))
                case pvz.PackageTypeBoxTape:
                        if order.Weight > 30 {
                                return errors.EntityWeightError("Box+Tape", strconv.FormatUint(uint64(order.OrderID), 10))
                        }
                        finalPrice = finalPrice.Add(decimal.NewFromInt(21))
                default:
                        return errors.EntityPackageError("Package Type", strconv.FormatUint(uint64(order.OrderID), 10))
                }
        }

        newOrder := &pvz.Order{
                OrderCreate: order,
                Status:      pvz.OrderStatusAccepted,
        }

        newOrder.Price = finalPrice

        return s.repo.SaveOrder(ctx, newOrder)
}

type OrderHistory struct {
        OrderID uint64
        Status  pvz.OrderStatus
        Time    time.Time
}

type OrderHistoryList struct {
        History []OrderHistory
}

func (s *Service) OrderHistory(ctx context.Context) (*OrderHistoryList, error) {
        rawHistory, err := s.repo.GetHistory(ctx)
        if err != nil {
                return nil, err
        }

        result := make([]OrderHistory, 0, len(rawHistory))
        for _, item := range rawHistory {
                result = append(result, OrderHistory{
                        OrderID: uint64(item.OrderID),
                        Status:  item.Status,
                        Time:    item.Time,
                })
        }

        sort.Slice(result, func(i, j int) bool {
                return result[i].Time.After(result[j].Time)
        })

        return &OrderHistoryList{History: result}, nil
}

func (s *Service) StatusHistory(ctx context.Context, orderID pvz.OrderID) ([]*pvz.StatusChange, error) {
        rawHistory, err := s.repo.GetStatusHistory(ctx, orderID)
        if err != nil {
                return nil, err
        }

        if len(rawHistory) == 0 {
                return nil, fmt.Errorf("order %d not found", orderID)
        }

        return rawHistory, nil
}

import (
        "context"
        "fmt"
        "gitlab.ozon.dev/gleb_evseev/homework/internal/domain/errors"
        "gitlab.ozon.dev/gleb_evseev/homework/internal/domain/pvz"

type ProcessOrdersInput struct {
        UserID   uint64
        Action   pvz.ActionType
        OrderIDs []uint64
}

type ProcessOrdersOutput struct {
        Processed []uint64
        Errors    []uint64
}

func (s *Service) Process(ctx context.Context, input ProcessOrdersInput) (ProcessOrdersOutput, error) {
        
        if input.Action != pvz.ActionTypeIssue && input.Action != pvz.ActionTypeReturn {
                return ProcessOrdersOutput{}, errors.EntityValidationError("Action", "unsupported action")
        }

        var (
                processed []uint64
                failed    []uint64
                now       = s.nowFn().UTC()
        )

        for _, id := range input.OrderIDs {
                order, err := s.repo.GetOrder(ctx, pvz.OrderID(id))
                if err != nil {
                        failed = append(failed, id)
                        continue
                }

                switch input.Action {
                case pvz.ActionTypeIssue:
                        if order.OrderCreate.ExpiresAt.Before(now) {
                                failed = append(failed, id)
                                continue
                        }
                        order.Status = pvz.OrderStatusIssued

                case pvz.ActionTypeReturn:
                        if order.IsExpiredForReturn(now) {
                                failed = append(failed, id)
                                continue
                        }
                        order.Status = pvz.OrderStatusReturned
                        order.OrderCreate.ExpiresAt = now

                default:
                        return ProcessOrdersOutput{}, errors.EntityValidationError("Action", fmt.Sprint(input.Action))
                }

                err = s.repo.UpdateOrder(ctx, order)
                if err != nil {
                        failed = append(failed, id)
                        continue
                }

                processed = append(processed, id)
        }

        return ProcessOrdersOutput{
                Processed: processed,
                Errors:    failed,
        }, nil
}

}

func (s *Service) Return(ctx context.Context, input ReturnOrderInput) error {
        order, err := s.repo.GetOrder(ctx, input.OrderID)
        if err != nil {
                return err
        }

        if order.Status == pvz.OrderStatusReturned {
                return errors.EntityValidationError("Order", strconv.FormatUint(uint64(order.OrderID), 10))
        }

        if order.Status == pvz.OrderStatusAccepted && order.ExpiresAt.Before(s.nowFn().UTC()) {
                return errors.EntityExpiredError("Order", strconv.FormatUint(uint64(order.OrderID), 10))
        }

        order.Status = pvz.OrderStatusReturned
        order.ExpiresAt = s.nowFn().UTC()

        err = s.repo.UpdateOrder(ctx, order)
        if err != nil {
                return fmt.Errorf("update order: %w", err)
        }

        return nil
}

func (s *Storage) GetOrder(ctx context.Context, orderID pvz.OrderID) (*pvz.Order, error) {
        const query = `
                SELECT order_id, user_id, status, expires_at, weight, price, package_type
                FROM orders
                WHERE order_id = $1
        `

        var (
                order   pvz.Order
                pkgType *int32
        )

        err := s.db.QueryRow(ctx, query, orderID).Scan(
                &order.OrderID,
                &order.UserID,
                &order.Status,
                &order.ExpiresAt,
                &order.Weight,
                &order.Price,
                &pkgType,
        )
        if err != nil {
                if errors.Is(err, pgx.ErrNoRows) {
                        return nil, storageErrors.EntityNotFoundError("Order", fmt.Sprint(orderID))
                }
                return nil, err
        }

        if pkgType != nil {
                pkg := pvz.PackageType(*pkgType)
                order.Package = &pkg
        }

        return &order, nil
}

func (s *Storage) ListOrders(ctx context.Context, options pvz.ListOptions) ([]*pvz.Order, int32, error) {
        var (
                whereClauses []string
                args         []interface{}
        )
        whereClauses = append(whereClauses, fmt.Sprintf("user_id = $%d", len(args)+1))
        args = append(args, options.UserID)

        if options.InPVZ {
                whereClauses = append(whereClauses, fmt.Sprintf("status = $%d", len(args)+1))
                args = append(args, pvz.OrderStatusAccepted)
        }

        whereSQL := "WHERE " + strings.Join(whereClauses, " AND ")

        var total int32
        countQuery := "SELECT COUNT(*) FROM orders " + whereSQL
        if err := s.db.QueryRow(ctx, countQuery, args...).Scan(&total); err != nil {
                log.Printf("ListOrders COUNT error: %v, query: %s, args: %v", err, countQuery, args)
                return nil, 0, err
        }

        var limit, offset *uint64
        if options.Pagination != nil {
                limit = options.Pagination.Limit
                if options.Pagination.Page != nil && limit != nil {
                        tmp := (*options.Pagination.Page) * (*limit)
                        offset = &tmp

                        if int32(*offset) >= total {
                                return []*pvz.Order{}, total, nil
                        }
                }
        }

        query := `
                SELECT order_id, user_id, status, expires_at, weight, price, package_type
                FROM orders
        ` + whereSQL + " ORDER BY order_id"

        if limit != nil {
                query += fmt.Sprintf(" LIMIT %d", *limit)
        }
        if offset != nil {
                query += fmt.Sprintf(" OFFSET %d", *offset)
        }

        rows, err := s.db.Query(ctx, query, args...)
        if err != nil {
                log.Printf("ListOrders SELECT error: %v, query: %s, args: %v", err, query, args)
                return nil, 0, err
        }
        defer rows.Close()

        var orders []*pvz.Order
        for rows.Next() {
                var o pvz.Order
                var pkgType *string
                if err := rows.Scan(
                        &o.OrderID,
                        &o.UserID,
                        &o.Status,
                        &o.ExpiresAt,
                        &o.Weight,
                        &o.Price,
                        &pkgType,
                ); err != nil {
                        return nil, 0, err
                }
                orders = append(orders, &o)
        }
        
        return orders, total, nil
}

func (s *Storage) GetHistory(ctx context.Context) ([]*pvz.StatusChange, error) {
        query := `
                SELECT order_id, status, changed_at
                FROM order_status_history
                ORDER BY changed_at DESC
        `

        rows, err := s.db.Query(ctx, query)
        if err != nil {
                return nil, err
        }
        defer rows.Close()

        var history []*pvz.StatusChange
        for rows.Next() {
                var h pvz.StatusChange
                if err := rows.Scan(&h.OrderID, &h.Status, &h.Time); err != nil {
                        return nil, err
                }
                history = append(history, &h)
        }

        return history, nil
}

func (s *Storage) GetStatusHistory(ctx context.Context, orderID pvz.OrderID) ([]*pvz.StatusChange, error) {
        const query = `
                SELECT status, changed_at
                FROM order_status_history
                WHERE order_id = $1
                ORDER BY changed_at DESC
        `

        rows, err := s.db.Query(ctx, query, orderID)
        if err != nil {
                return nil, err
        }
        defer rows.Close()

        var history []*pvz.StatusChange
        for rows.Next() {
                var status int
                var changedAt time.Time
                if err := rows.Scan(&status, &changedAt); err != nil {
                        return nil, err
                }
                history = append(history, &pvz.StatusChange{
                        OrderID: orderID,
                        Status:  pvz.OrderStatus(status),
                        Time:    changedAt,
                })
        }

        if err := rows.Err(); err != nil {
                return nil, err
        }

        if len(history) == 0 {
                return nil, storageErrors.EntityNotFoundError("Order", fmt.Sprint(orderID))
        }

        return history, nil
}

func (s *Storage) SaveOrder(ctx context.Context, order *pvz.Order) error {
        tx, err := s.db.Begin(ctx)
        if err != nil {
                return err
        }
        defer func() {
                if err != nil {
                        _ = tx.Rollback(ctx)
                }
        }()

        const insertOrder = `
                INSERT INTO orders (order_id, user_id, status, expires_at, weight, price, package_type)
                VALUES ($1, $2, $3, $4, $5, $6, $7)
        `

        _, err = tx.Exec(ctx, insertOrder,
                order.OrderID,
                order.UserID,
                order.Status,
                order.ExpiresAt,
                order.Weight,
                order.Price,
                toDBPackageType(order.Package),
        )
        if err != nil {
                return err
        }

        const insertHistory = `
                INSERT INTO order_status_history (order_id, status, changed_at)
                VALUES ($1, $2, NOW())
        `

        _, err = tx.Exec(ctx, insertHistory, order.OrderID, order.Status)
        if err != nil {
                return err
        }

        err = tx.Commit(ctx)
        return err
}

func (s *Storage) UpdateOrder(ctx context.Context, order *pvz.Order) (err error) {
        tx, err := s.db.Begin(ctx)
        if err != nil {
                return err
        }
        defer func() {
                if err != nil {
                        _ = tx.Rollback(ctx)
                }
        }()

        const updateOrder = `
                UPDATE orders SET status = $1, expires_at = $2, weight = $3, price = $4, package_type = $5
                WHERE order_id = $6
        `
        commandTag, err := tx.Exec(ctx, updateOrder,
                order.Status,
                order.ExpiresAt,
                order.Weight,
                order.Price,
                order.Package,
                order.OrderID,
        )
        if err != nil {
                return err
        }

        if commandTag.RowsAffected() == 0 {
                return storageErrors.EntityNotFoundError("Order", fmt.Sprint(order.OrderID))
        }

        const insertHistory = `
                INSERT INTO order_status_history (order_id, status, changed_at)
                VALUES ($1, $2, NOW())
        `
        _, err = tx.Exec(ctx, insertHistory, order.OrderID, order.Status)
        if err != nil {
                return err
        }

        err = tx.Commit(ctx)
        return err
}
